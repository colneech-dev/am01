# Raspberry Pi firmware config for the AM01 miner (CM4, 64-bit).
# Installed to the boot partition as config.txt.

start_file=start4.elf
fixup_file=fixup4.dat

kernel=Image
arm_64bit=1

# NO UART console. The manual's GPIO table (section 2.2.9) routes GPIO14 -> C17
# and GPIO15 -> A15 into the FPGA, and those are the primary UART's TXD/RXD as
# well as data bus bits 14 and 15. Every alternate CM4 UART (uart2..uart5) also
# lands inside GPIO0-27, so there is no serial console available on the pins.
#
# Serial console is provided over the mini-USB (J14) instead, as a USB CDC-ACM
# gadget -- see /etc/modules-load.d/am01-usb-gadget.conf. It appears as a COM
# port on the machine at the other end of the cable.
#
# BOTH the USB-A ports and the gadget console are available, selected
# automatically by whether a mini-USB cable is plugged in. VERIFIED ON HARDWARE
# 2026-08-29: unplugged, the keyboard works; plugged, a COM port appears.
#
# How it works. Schematic sheet 1: the FSUSB42MUX (U15) has its common port on
# the CM4's USB2_P/USB2_N, output 1 (HSD1) on the USB2514 hub feeding the USB-A
# ports, output 2 (HSD2) on the mini-USB J14. SEL is USBOTG_ID -- the OTG ID
# pin -- so the mux follows cable presence:
#
#   mini-USB unplugged -> ID low  -> mux to HSD1 (hub)      -> role: host
#   mini-USB plugged   -> ID high -> mux to HSD2 (mini-USB) -> role: peripheral
#
# dr_mode=otg makes the dwc2 driver take its role from that same ID pin, so the
# software role and the mux track each other with no configuration at all.
#
# CORRECTION. Earlier revisions of this file asserted the opposite -- that the
# two were "mutually exclusive and cannot both be had", that R278 pulls ID
# permanently HIGH, and that "dr_mode=otg does NOT split the difference ... otg
# resolves to peripheral and the USB-A ports stay dead". That was wrong, and it
# was wrong in a way the board itself disproved: if ID were statically high the
# mux would sit on the mini-USB permanently and the USB-A ports would never
# work, yet a keyboard works whenever the mini-USB is unplugged. The claim came
# from reading a pull-up off the schematic without checking it against the
# board's actual behaviour.
#
# NOT otg_mode=1. That routes the OTG port through the xHCI controller, which
# is host-only -- it pins the role and defeats the automatic switching above.
# The cost of dropping it is that host duties fall to dwc2 rather than xHCI,
# which is the slower path; fine for a keyboard behind the hub, which is all
# this board needs locally. rpiboot is unaffected either way -- it runs before
# Linux.
dtoverlay=dwc2,dr_mode=otg

# The FPGA parallel bus claims GPIO 0-23, so nothing else may drive them.
# Disable the I2C/SPI peripherals that would otherwise take those pins.
dtparam=i2c_arm=off
dtparam=spi=off
dtparam=i2s=off

# Reserve the bus pins from other drivers.
dtoverlay=am01-fpga-gpio

# No display attached by default; odo-ui is skipped when /dev/fb0 is absent.
# Raise this and attach a screen if you want the touch UI.
gpu_mem=64

# odo-ui requires an RGB565 framebuffer -- it exits immediately with
# "expected 16bpp framebuffer, got 32" against the default 32bpp one. RGB565 is
# also the ILI9341 panel s native format, which is why the UI was written for it.
framebuffer_depth=16
# 16bpp with an alpha channel confuses the firmware s format selection; ignoring
# alpha keeps it at a plain RGB565 surface.
framebuffer_ignore_alpha=1

disable_overscan=1
