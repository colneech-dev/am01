# Raspberry Pi firmware config for the AM01 miner (CM4, 64-bit).
# Installed to the boot partition as config.txt.

start_file=start4.elf
fixup_file=fixup4.dat

kernel=Image
arm_64bit=1

# NO UART console. The manual's GPIO table (section 2.2.9) routes GPIO14 -> C17
# and GPIO15 -> A15 into the FPGA, and those are the primary UART's TXD/RXD as
# well as data bus bits 14 and 15. Every alternate CM4 UART (uart2..uart5) also
# lands inside GPIO0-27, so there is no serial console available on the pins.
#
# Serial console is provided over the mini-USB (J14) instead, as a USB CDC-ACM
# gadget -- see /etc/modules-load.d/am01-usb-gadget.conf. It appears as a COM
# port on the machine at the other end of the cable.
#
# HOST mode: the four USB-A ports work, and there is NO gadget console on J14.
#
# These are mutually exclusive on this board and cannot both be had. Schematic
# sheet 1: the FSUSB42MUX (U15) has its common port on the CM4's USB2_P/USB2_N,
# output 1 (HSD1) on the USB2514 hub feeding the USB-A ports, output 2 (HSD2)
# on the mini-USB J14. SEL is USBOTG_ID, pulled HIGH by R278 -- and per the
# FSUSB42 truth table SEL high selects HSD2, the mini-USB. Peripheral mode also
# means no VBUS on the host ports, so a keyboard gets no power.
#
# dr_mode=otg does NOT split the difference: a pulled-up ID reads as
# "B-device", so otg resolves to peripheral and the USB-A ports stay dead.
# Host mode is what drives USBOTG_ID low, flipping the mux to the hub.
#
# Chosen because HDMI + USB keyboard is the only local console when there is
# no network: a login prompt you cannot type at is no use. Switch back to the
# gadget (dtoverlay=dwc2,dr_mode=peripheral, and drop otg_mode) once the board
# is reachable over the network. rpiboot is unaffected either way -- it runs
# before Linux.
#
# otg_mode=1, NOT dtoverlay=dwc2,dr_mode=host. On the CM4 the USB 2.0 host
# controller is disabled by default, and dr_mode=host does not enable it --
# it only tells the dwc2 driver which role to take. otg_mode=1 routes the OTG
# port through the xHCI controller instead, which is what actually powers and
# enumerates the USB-A ports behind the USB2514 hub. The two are alternatives:
# do not set both.
otg_mode=1

# The FPGA parallel bus claims GPIO 0-23, so nothing else may drive them.
# Disable the I2C/SPI peripherals that would otherwise take those pins.
dtparam=i2c_arm=off
dtparam=spi=off
dtparam=i2s=off

# Reserve the bus pins from other drivers.
dtoverlay=am01-fpga-gpio

# No display attached by default; odo-ui is skipped when /dev/fb0 is absent.
# Raise this and attach a screen if you want the touch UI.
gpu_mem=64

disable_overscan=1
